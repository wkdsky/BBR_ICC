# 严格按竞争程度 rho 配置：rho = (flows * sender_bw) / bottle_bw
# 覆盖：
#   flows = 4,8,16,32
#   rho   = 0.8,1.6,3.2
#   queue-bdp = 1,4,10
#   bottle-bw = 20,100 Mbps
# 固定：
#   bottle-delay = 18ms  (RTT约40ms)
#   sim-time = 120s
#
# 注：sender-bw 由 rho 反推，部分组合会出现小数（如 32流@20Mbps -> 0.5/1/2）。
# run_bbrv2_batch.sh 已支持小数 Mbps。

# ===== bottle-bw=20Mbps =====
# --- flows=4 ---
# rho=0.8 => sender-bw=4Mbps
--flows 4 --sender-bw 4 --bottle-bw 20 --bottle-delay 18 --queue-bdp 1 --sim-time 120
--flows 4 --sender-bw 4 --bottle-bw 20 --bottle-delay 18 --queue-bdp 4 --sim-time 120
--flows 4 --sender-bw 4 --bottle-bw 20 --bottle-delay 18 --queue-bdp 10 --sim-time 120
# rho=1.6 => sender-bw=8Mbps
--flows 4 --sender-bw 8 --bottle-bw 20 --bottle-delay 18 --queue-bdp 1 --sim-time 120
--flows 4 --sender-bw 8 --bottle-bw 20 --bottle-delay 18 --queue-bdp 4 --sim-time 120
--flows 4 --sender-bw 8 --bottle-bw 20 --bottle-delay 18 --queue-bdp 10 --sim-time 120
# rho=3.2 => sender-bw=16Mbps
--flows 4 --sender-bw 16 --bottle-bw 20 --bottle-delay 18 --queue-bdp 1 --sim-time 120
